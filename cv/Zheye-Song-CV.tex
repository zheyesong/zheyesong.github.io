    \resumeProjectHeading
        {Causal Inference in Metric Spaces via Distance Profiles}
        {2025.08 -- 2026.05}
        {Under Review}
        {Advisor: Zhiheng Zhang, Tenure-Track Assistant Professor, SUFE}
        {}
        \resumeItemListStart
            \resumeItem{Developed a distance-profile causal inference framework for object-valued outcomes by transforming metric-ball events in separable metric spaces into geometry-indexed binary potential-outcome contrasts.}
            \resumeItem{Proved pointwise identification, doubly robust/AIPW estimation properties, Neyman orthogonality, and auxiliary-stabilized finite-grid simultaneous inference under causal identification and high-level nuisance conditions.}
            \resumeItem{Validated the framework across Euclidean, Wasserstein, SPD, graph, and tree-valued outcomes, with semi-synthetic connectome and network benchmarks supporting calibrated coverage, global testing, and robustness diagnostics.}
        \resumeItemListEnd

    \resumeProjectHeading

        {Advisor: Jiaxi Yang, Alumni Researcher, Columbia University}
        {}
        \resumeItemListStart

            \resumeItem{Implemented HybridFull, an overlap-aware orthogonal-residual stacking rule that learns simplex weights over candidate CATE estimators within propensity strata, and evaluated it across S1--S6 simulations plus IHDP/JOBS benchmarks.}
